function doGet(e) {
  var action = e.parameter.action;
  var sheetName = e.parameter.sheetName || "Sheet1";

  if (action === "getSaveStatus") {
    var requestId = String(e.parameter.requestId || "");
    if (!/^[a-zA-Z0-9-]{10,80}$/.test(requestId)) {
      return ContentService.createTextOutput(JSON.stringify({ status: "invalid" }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    var savedStatus = CacheService.getScriptCache().get("saveStatus:" + requestId);
    return ContentService.createTextOutput(savedStatus || JSON.stringify({ status: "pending" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // Khóa bảng học sinh cho tới khi xác thực được người gọi ở phía máy chủ.
  if (action === "getLeaderboard") {
    return ContentService.createTextOutput(JSON.stringify({ status: "auth_required" }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  // 1. API Lấy từ vựng: ?action=getVocab&sheetName=Unit1
  if (action === "getVocab") {
    // Không cho endpoint từ vựng công khai đọc các sheet lưu kết quả/góp ý.
    var restrictedSheets = ["scores", "details", "feedback"];
    if (restrictedSheets.indexOf(String(sheetName).trim().toLowerCase()) !== -1) {
      return ContentService.createTextOutput(JSON.stringify([]))
        .setMimeType(ContentService.MimeType.JSON);
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(sheetName);
    
    if (!sheet) {
      return ContentService.createTextOutput(JSON.stringify([]))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    var data = sheet.getDataRange().getValues();
    var headers = data[0]; // Giả sử dòng 1 là header: word, ipa, level, type, meaning, context
    var vocabList = [];
    
    for (var i = 1; i < data.length; i++) {
      if (data[i][0] !== "") { // Nếu cột word không rỗng
        vocabList.push({
          word: String(data[i][0]),
          ipa: String(data[i][1] || ""),
          level: String(data[i][2] || ""),
          type: String(data[i][3] || ""),
          meaning: String(data[i][4] || ""),
          context: String(data[i][5] || "")
        });
      }
    }
    
    return ContentService.createTextOutput(JSON.stringify(vocabList))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  var data = null;
  var requestId = "";
  try {
    data = JSON.parse(e.postData.contents);
    requestId = String(data.requestId || "");
    var action = data.action;
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // 1. Gửi góp ý ẩn danh từ panel góp ý công khai.
    if (action === "submitFeedback") {
      // Honeypot chống bot đơn giản; bỏ qua yêu cầu có điền ô ẩn.
      if (String(data.website || "").trim() !== "") {
        return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
          .setMimeType(ContentService.MimeType.JSON);
      }

      var categories = ["Tính năng", "Nội dung", "Giao diện", "Lỗi", "Khác"];
      var category = String(data.category || "").trim();
      var feedback = String(data.message || "").trim();
      if (categories.indexOf(category) === -1 || feedback.length < 10 || feedback.length > 2000) {
        return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Invalid feedback" }))
          .setMimeType(ContentService.MimeType.JSON);
      }

      var feedbackSheet = ss.getSheetByName("Feedback") || ss.insertSheet("Feedback");
      if (feedbackSheet.getLastRow() === 0) {
        feedbackSheet.appendRow(["Timestamp", "Chủ đề", "Góp ý"]);
      }
      feedbackSheet.appendRow([new Date(), category, safeFeedbackCell(feedback)]);
    }
    // 2. API Lưu điểm tổng
    else if (action === "saveScore") {
      var submittedMode = data.mode === "free" ? "free" : "guided";
      var submittedName = String(data.name || "").trim();
      // Chế độ tự học không được gửi kết quả lên Apps Script; chỉ chấp nhận ID học sinh đã đăng ký.
      if (submittedMode !== "guided" || !isGuidedStudentId(submittedName)) {
        setSaveStatus(requestId, "ignored");
        return ContentService.createTextOutput(JSON.stringify({ status: "ignored" }))
          .setMimeType(ContentService.MimeType.JSON);
      }
      var sheet = ss.getSheetByName("Scores") || ss.insertSheet("Scores");
      // Nếu chưa có header thì tạo header
      if (sheet.getLastRow() === 0) {
        sheet.appendRow(["Timestamp", "Tên học sinh", "Bài học", "Game Type", "Điểm đạt", "Tổng điểm", "Thời lượng (giây)", "Chế độ"]);
      }
      var submittedDuration = Math.max(0, Math.floor(Number(data.duration) || 0));
      sheet.appendRow([
        new Date(),
        safeFeedbackCell(submittedName.slice(0, 40)),
        safeFeedbackCell(String(data.sheetName || "").slice(0, 100)),
        safeFeedbackCell(String(data.gameType || "Speech Reflex").slice(0, 80)),
        Number(data.score) || 0,
        Number(data.total) || 0,
        submittedDuration,
        submittedMode
      ]);
      CacheService.getScriptCache().remove("guidedLeaderboard");
      try {
        UrlFetchApp.fetch("https://script.google.com/macros/s/AKfycbwJ3d4lCV4MS1Ks6HuGaIPQfv-X6Gn0c5UV3WV-mxge9NUgt1vCkEEJcXFx9YwOEg_S/exec", { method: "post" });
      } catch (syncError) {
        console.warn("Đã lưu điểm nhưng đồng bộ phụ thất bại:", syncError);
      }
      setSaveStatus(requestId, "score_saved");

    } 
    // 3. API Lưu chi tiết từng câu
    else if (action === "saveDetailList") {
      if (!isGuidedStudentId(data.name)) {
        setSaveStatus(requestId, "ignored");
        return ContentService.createTextOutput(JSON.stringify({ status: "ignored" }))
          .setMimeType(ContentService.MimeType.JSON);
      }
      var sheet = ss.getSheetByName("Details") || ss.insertSheet("Details");
      if (sheet.getLastRow() === 0) {
        sheet.appendRow(["Timestamp", "Tên học sinh", "Bài học", "Ngữ cảnh câu", "Từ chuẩn", "Học sinh trả lời", "Đúng/Sai", "Điểm", "Lịch sử đọc sai (STT)"]);
      }
      
      var time = new Date();
      var details = data.details || [];
      for (var i = 0; i < details.length; i++) {
        sheet.appendRow([
          time,
          data.name,
          data.sheetName,
          details[i].question,
          details[i].targetWord,
          details[i].studentAnswer,
          details[i].isCorrect ? "Đúng" : "Sai",
          details[i].points,
          details[i].wrongAttempts || "Không có"
        ]);
      }
      setSaveStatus(requestId, "saved");
    }
    else {
      throw new Error("Unsupported action");
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (error) {
    setSaveStatus(requestId, "error", error.toString());
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function setSaveStatus(requestId, status, message) {
  if (!/^[a-zA-Z0-9-]{10,80}$/.test(String(requestId || ""))) return;
  var result = { status: status };
  if (message) result.message = String(message).slice(0, 180);
  CacheService.getScriptCache().put("saveStatus:" + requestId, JSON.stringify(result), 300);
}

// Tránh hiểu nội dung góp ý bắt đầu bằng ký hiệu công thức thành công thức trong Sheets.
function safeFeedbackCell(value) {
  var text = String(value || "");
  return /^[=+\-@]/.test(text) ? "'" + text : text;
}

function isGuidedStudentId(value) {
  var allowedIds = ["trinh", "duong gia dat", "minh", "vy"];
  var normalized = String(value || "").trim().toLowerCase().replace(/\s+/g, " ");
  return allowedIds.indexOf(normalized) !== -1;
}
